function cityColor(c) { return CITY_COLORS[c] || '#5a635a'; }

function statusBadge(s) {
  const cfg = {
    pending: { cls: 'badge-pending', icon: clockIcon(), label: 'Pending' },
    flagged:  { cls: 'badge-flagged',  icon: flagIcon(),  label: 'Flagged' },
    dup:      { cls: 'badge-dup',      icon: copyIcon(),  label: 'Possible dup' },
    approved: { cls: 'badge-approved', icon: checkIcon(), label: 'Approved' },
    rejected: { cls: 'badge-rejected', icon: xIcon(),     label: 'Rejected' },
  };
  const c = cfg[s] || cfg.pending;
  return `<span class="badge ${c.cls}">${c.icon} ${c.label}</span>`;
}

function methodBadge(m) {
  return m === 'OTP'
    ? `<span class="badge badge-otp">${shieldIcon()} OTP verified</span>`
    : `<span class="badge badge-pending">${fileIcon()} Document</span>`;
}

function stars(n) {
  return `<span class="stars">${'★'.repeat(n)}</span><span class="stars-dim">${'★'.repeat(5 - n)}</span>`;
}

function toast(msg, type = 'green') {
  const icon = type === 'green' ? checkIcon() : type === 'red' ? xIcon() : warnIcon();
  const el = document.createElement('div');
  el.className = `toast ${type}`;
  el.innerHTML = `${icon} ${msg}`;
  document.getElementById('toasts').appendChild(el);
  setTimeout(() => el.remove(), 3500);
}

function getFilteredQueue() {
  const { search, city, category, tab } = State.queueFilter;
  return DATA.queue.filter(r => {
    if (tab === 'dup' && r.status !== 'dup') return false;
    if (tab === 'flagged' && r.status !== 'flagged') return false;
    if (search && !`${r.name} ${r.address}`.toLowerCase().includes(search.toLowerCase())) return false;
    if (city && r.city !== city) return false;
    if (category && r.category !== category) return false;
    return true;
  });
}

// ─── ACTIONS ──────────────────────────────────────────────────────────────────

function quickAction(id, action) {
  const idx = DATA.queue.findIndex(r => r.id === id);
  if (idx < 0) return;
  DATA.queue.splice(idx, 1);
  if (action === 'approve') {
    State.stats.approved++;
    State.stats.pending = Math.max(0, State.stats.pending - 1);
    toast('Listing approved and published.', 'green');
  } else {
    State.stats.pending = Math.max(0, State.stats.pending - 1);
    toast('Listing rejected.', 'amber');
  }
  updateStats();
  renderQueue();
  closeModal();
}

function approveEdit(id) {
  const idx = DATA.edits.findIndex(e => e.id === id);
  if (idx >= 0) DATA.edits.splice(idx, 1);
  State.stats.edits = Math.max(0, State.stats.edits - 1);
  updateStats();
  renderEdits();
  toast('Edit approved and applied.', 'green');
}

function rejectEdit(id) {
  const idx = DATA.edits.findIndex(e => e.id === id);
  if (idx >= 0) DATA.edits.splice(idx, 1);
  State.stats.edits = Math.max(0, State.stats.edits - 1);
  updateStats();
  renderEdits();
  toast('Edit rejected.', 'amber');
}

function approveClaim(id) {
  const idx = DATA.claims.findIndex(c => c.id === id);
  if (idx >= 0) DATA.claims.splice(idx, 1);
  State.stats.claims = Math.max(0, State.stats.claims - 1);
  updateStats();
  renderClaims();
  toast('Claim approved — business verified!', 'green');
  closeModal();
}

function rejectClaim(id) {
  const idx = DATA.claims.findIndex(c => c.id === id);
  if (idx >= 0) DATA.claims.splice(idx, 1);
  State.stats.claims = Math.max(0, State.stats.claims - 1);
  updateStats();

  toast('Claim rejected.', 'amber');
  closeModal();
}

function keepReview(id) {

  if (idx >= 0) DATA.reviews.splice(idx, 1);
  renderReviews();
  toast('Review kept — flags cleared.', 'green');
}

function removeReview(id) {
  const idx = DATA.reviews.findIndex(r => r.id === id);
  if (idx >= 0) DATA.reviews.splice(idx, 1);
  renderReviews();
  toast('Review removed.', 'amber');
}

// ─── BULK ACTIONS ─────────────────────────────────────────────────────────────

function toggleSelectAll(cb) {
  document.querySelectorAll('.row-cb').forEach(c => {
    c.checked = cb.checked;
    cb.checked ? State.selected.add(parseInt(c.dataset.id)) : State.selected.delete(parseInt(c.dataset.id));
  });
  updateBulk();
}

function toggleRow(cb) {
  const id = parseInt(cb.dataset.id);
  cb.checked ? State.selected.add(id) : State.selected.delete(id);
  updateBulk();
}

function updateBulk() {
  const n = State.selected.size;
  const bar = document.getElementById('bulk-bar');
  bar.classList.toggle('visible', n > 0);
  document.getElementById('bulk-count').textContent = n;
  document.getElementById('bulk-approve-btn').style.display = n > 0 ? '' : 'none';
  document.getElementById('bulk-reject-btn').style.display = n > 0 ? '' : 'none';
}

function bulkAction(action) {
  const ids = [...State.selected];
  ids.forEach(id => {
    const idx = DATA.queue.findIndex(r => r.id === id);
    if (idx >= 0) DATA.queue.splice(idx, 1);
  });
  if (action === 'approve') {
    State.stats.approved += ids.length;
    toast(`${ids.length} listing${ids.length > 1 ? 's' : ''} approved.`, 'green');
  } else {
    toast(`${ids.length} listing${ids.length > 1 ? 's' : ''} rejected.`, 'amber');
  }
  State.stats.pending = Math.max(0, State.stats.pending - ids.length);
  State.selected.clear();
  updateStats();
  updateBulk();
  document.getElementById('select-all').checked = false;
  renderQueue();
}
