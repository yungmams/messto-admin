// ── BADGE HELPERS ─────────────────────────────────────────────────────────────
function statusBadge(s){
  var m={pending:{cls:'badge-pending',ic:IC.clock,lbl:'Pending'},flagged:{cls:'badge-flagged',ic:IC.flag,lbl:'Flagged'},dup:{cls:'badge-dup',ic:IC.copy,lbl:'Possible dup'},approved:{cls:'badge-approved',ic:IC.check,lbl:'Approved'},rejected:{cls:'badge-rejected',ic:IC.x,lbl:'Rejected'}};
  var c=m[s]||m.pending;
  return '<span class="badge '+c.cls+'">'+c.ic+' '+c.lbl+'</span>';
}
function methodBadge(m){return m==='OTP'?'<span class="badge badge-otp">'+IC.shield+' OTP verified</span>':'<span class="badge badge-pending">'+IC.file+' Document</span>';}
function stars(n){var s='';for(var i=0;i<5;i++)s+=i<n?'<span class="stars">★</span>':'<span class="stars-dim">★</span>';return s;}

// ── TOAST ─────────────────────────────────────────────────────────────────────
function toast(msg,type){
  type=type||'green';
  var ic=type==='green'?IC.check:type==='red'?IC.x:IC.warn;
  var el=document.createElement('div');
  el.className='toast '+type;
  el.innerHTML=ic+' '+msg;
  document.getElementById('toasts').appendChild(el);
  setTimeout(function(){el.remove();},3500);
}

// ── RENDER QUEUE ──────────────────────────────────────────────────────────────
function getFiltered(){
  return QUEUE.filter(function(r){
    if(qFilter.tab==='dup'&&r.status!=='dup')return false;
    if(qFilter.tab==='flagged'&&r.status!=='flagged')return false;
    if(qFilter.search&&(r.name+' '+r.address).toLowerCase().indexOf(qFilter.search.toLowerCase())<0)return false;
    if(qFilter.city&&r.city!==qFilter.city)return false;
    if(qFilter.cat&&r.category!==qFilter.cat)return false;
    return true;
  });
}

// ── ACTIONS ───────────────────────────────────────────────────────────────────
function quickAction(id,action){
  var idx=QUEUE.findIndex(function(r){return r.id===id;});
  if(idx>=0)QUEUE.splice(idx,1);
  if(action==='approve'){stats.approved++;stats.pending=Math.max(0,stats.pending-1);toast('Listing approved and published.','green');}
  else{stats.pending=Math.max(0,stats.pending-1);toast('Listing rejected.','amber');}
  updateStats();renderQueue();closeModal();
}
function approveEdit(id){var idx=EDITS.findIndex(function(e){return e.id===id;});if(idx>=0)EDITS.splice(idx,1);stats.edits=Math.max(0,stats.edits-1);updateStats();renderEdits();toast('Edit approved and applied.','green');}
function rejectEdit(id){var idx=EDITS.findIndex(function(e){return e.id===id;});if(idx>=0)EDITS.splice(idx,1);stats.edits=Math.max(0,stats.edits-1);updateStats();renderEdits();toast('Edit rejected.','amber');}
function approveClaim(id){var idx=CLAIMS.findIndex(function(c){return c.id===id;});if(idx>=0)CLAIMS.splice(idx,1);stats.claims=Math.max(0,stats.claims-1);updateStats();renderClaims();toast('Claim approved — business verified!','green');closeModal();}
function rejectClaim(id){var idx=CLAIMS.findIndex(function(c){return c.id===id;});if(idx>=0)CLAIMS.splice(idx,1);stats.claims=Math.max(0,stats.claims-1);updateStats();renderClaims();toast('Claim rejected.','amber');closeModal();}
function keepReview(id){var idx=REVIEWS.findIndex(function(r){return r.id===id;});if(idx>=0)REVIEWS.splice(idx,1);renderReviews();toast('Review kept — flags cleared.','green');}
function removeReview(id){var idx=REVIEWS.findIndex(function(r){return r.id===id;});if(idx>=0)REVIEWS.splice(idx,1);renderReviews();toast('Review removed.','amber');}

// ── BULK ──────────────────────────────────────────────────────────────────────
function toggleSelectAll(cb){
  document.querySelectorAll('.row-cb').forEach(function(c){
    c.checked=cb.checked;
    var id=parseInt(c.dataset.id);
    if(cb.checked){if(selected.indexOf(id)<0)selected.push(id);}else{selected=selected.filter(function(x){return x!==id;});}
  });
  updateBulk();
}
function toggleRow(cb){
  var id=parseInt(cb.dataset.id);
  if(cb.checked){if(selected.indexOf(id)<0)selected.push(id);}else{selected=selected.filter(function(x){return x!==id;});}
  updateBulk();
}
function updateBulk(){
  var n=selected.length;
  document.getElementById('bulk-bar').classList.toggle('visible',n>0);
  document.getElementById('bulk-count').textContent=n;
  document.getElementById('bulk-approve-btn').style.display=n>0?'':'none';
  document.getElementById('bulk-reject-btn').style.display=n>0?'':'none';
}
function bulkAction(action){
  var n=selected.length;
  selected.forEach(function(id){var idx=QUEUE.findIndex(function(r){return r.id===id;});if(idx>=0)QUEUE.splice(idx,1);});
  if(action==='approve'){stats.approved+=n;toast(n+' listing'+(n>1?'s':'')+' approved.','green');}
  else{toast(n+' listing'+(n>1?'s':'')+' rejected.','amber');}
  stats.pending=Math.max(0,stats.pending-n);
  selected=[];updateStats();updateBulk();
  document.getElementById('select-all').checked=false;
  renderQueue();
}
